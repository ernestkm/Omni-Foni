# SIP edge (Kamailio) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Put a real Kamailio SIP listener in front of the existing switching policy/charging chain so SIP INVITEs are authorized, relayed to an LCR-resolved trunk, billed, and torn down when the prepaid balance runs out.

**Architecture:** Kamailio owns SIP signaling and dialog state. A new Go service `sipedge` is the policy adapter: Kamailio calls it over HTTP on INVITE and BYE, it calls switching's new HTTP facade (`start`/`stop`/`live`), and a 1-second reaper diffs switching live calls against tracked dialogs, ending orphaned dialogs via Kamailio's `dlg.end_dlg` JSON-RPC. Routing gains a per-trunk `target_uri` so one policy round-trip returns both the decision and the relay destination.

**Tech Stack:** Go 1.27, Kamailio 5.8 (`ghcr.io/kamailio/kamailio:5.8.3-bookworm`), gRPC/HTTP, Postgres 17, Kafka, OpenTelemetry (shared `observability` module).

## Global Constraints

- Clean-room: never read ASTPP source, schemas, or licensed assets. Feature claims come only from the four marketing pages cited in `docs/master-plan.md` sections 0 and 8.
- FOSS-only stack. No Terraform, no Redis (use Valkey).
- Money is integer minor units (1/10000 currency); per-event cost is `ROUND(seconds*rate)`.
- Fail-closed: any control-plane outage denies the call (`5xx`), never allows it.
- Env names are exact: `SWITCHING_HTTP_ADDR`, `SIPEDGE_SHIM_ADDR`, `SIPEDGE_SWITCHING_HTTP`, `SIPEDGE_KAMAILIO_RPC`, `SIPEDGE_TRUNK_ACCOUNTS`, `SIPEDGE_REAP_INTERVAL`.
- Do not hand-edit generated protobuf code. Regenerate with `make protoc` (needs `protoc`, `protoc-gen-go`, `protoc-gen-go-grpc` in `/root/go/bin`).
- `make up` never rebuilds images: after code changes run `docker compose build <service>` before `make up`.
- Host tests need the dev stack up (`make up`) and the `/etc/hosts` entry `127.0.0.1 kafka` (already present).
- Work on a feature branch: `git checkout -b p4-sip-edge` from `master` first.

```go
	if req.TrunkId == "" {
```

```go
package main

import (
	"encoding/json"
	"net/http"
	"time"

	"google.golang.org/grpc"

	pb "omnifoni/switching/gen/switchingpb"
)

type startReq struct {
	AccountID   string `json:"account_id"`
	Destination string `json:"destination"`
	CLI         string `json:"cli"`
	SourceIP    string `json:"source_ip"`
}

type startResp struct {
	Allowed    bool   `json:"allowed"`
	Reason     string `json:"reason,omitempty"`
	CallID     string `json:"call_id,omitempty"`
	SessionID  string `json:"session_id,omitempty"`
	MaxSeconds int64  `json:"max_seconds,omitempty"`
	TargetURI  string `json:"target_uri,omitempty"`
}

type liveCall struct {
	CallID         string `json:"call_id"`
	SessionID      string `json:"session_id"`
	AccountID      string `json:"account_id"`
	Destination    string `json:"destination"`
	CLI            string `json:"cli"`
	TrunkID        string `json:"trunk_id"`
	StartedAtMs    int64  `json:"started_at_ms"`
	ElapsedSeconds int64  `json:"elapsed_seconds"`
	MaxSeconds     int64  `json:"max_seconds"`
	ChargedSeconds int64  `json:"charged_seconds"`
}

// newFacade exposes the call registry to the SIP edge over HTTP.
func newFacade(reg *Registry, cc *grpc.ClientConn) http.Handler {
	mux := http.NewServeMux()
	mux.HandleFunc("POST /start", func(w http.ResponseWriter, r *http.Request) {
		var req startReq
		if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
			http.Error(w, "bad request", http.StatusBadRequest)
			return
		}
		if req.AccountID == "" || req.Destination == "" {
			http.Error(w, "account_id and destination required", http.StatusBadRequest)
			return
		}
		res, err := reg.StartCall(r.Context(), cc, &pb.StartCallRequest{
			AccountId: req.AccountID, Destination: req.Destination,
			Cli: req.CLI, SourceIp: req.SourceIP,
		})
		if err != nil {
			http.Error(w, "internal error", http.StatusInternalServerError)
			return
		}
		writeJSON(w, startResp{
			Allowed: res.GetAllowed(), Reason: res.GetReason(),
			CallID: res.GetCallId(), SessionID: res.GetSessionId(),
			MaxSeconds: res.GetMaxSeconds(), TargetURI: res.GetTargetUri(),
		})
	})
	mux.HandleFunc("POST /stop", func(w http.ResponseWriter, r *http.Request) {
		var req struct {
			CallID string `json:"call_id"`
		}
		if err := json.NewDecoder(r.Body).Decode(&req); err != nil || req.CallID == "" {
			http.Error(w, "bad request", http.StatusBadRequest)
			return
		}
		writeJSON(w, map[string]bool{"stopped": reg.StopCall(req.CallID)})
	})
	mux.HandleFunc("GET /live", func(w http.ResponseWriter, _ *http.Request) {
		snaps := reg.LiveCalls(nil)
		now := time.Now()
		out := make([]liveCall, 0, len(snaps))
		for _, c := range snaps {
			elapsed := int64(now.Sub(c.StartedAt).Seconds())
			if elapsed < 0 {
				elapsed = 0
			}
			out = append(out, liveCall{
				CallID: c.CallID, SessionID: c.SessionID, AccountID: c.AccountID,
				Destination: c.Destination, CLI: c.CLI, TrunkID: c.TrunkID,
				StartedAtMs: c.StartedAt.UnixMilli(), ElapsedSeconds: elapsed,
				MaxSeconds: c.MaxSeconds, ChargedSeconds: c.ChargedSeconds,
			})
		}
		writeJSON(w, map[string]any{"calls": out})
```

```go
package main

import (
	"net/http"
	"strings"
)

// parseAccounts parses "ip=account,ip=account" into a lookup table.
func parseAccounts(s string) map[string]string {
	m := map[string]string{}
	for _, pair := range strings.Split(s, ",") {
		pair = strings.TrimSpace(pair)
		if pair == "" {
			continue
		}
		ip, acct, ok := strings.Cut(pair, "=")
		if !ok {
			continue
		}
		ip, acct = strings.TrimSpace(ip), strings.TrimSpace(acct)
		if ip != "" && acct != "" {
			m[ip] = acct
		}
	}
	return m
}

// sipCode maps a switching deny reason to an HTTP status the SIP edge turns
// into a SIP reply: policy denials are 403, control-plane outages are 503.
func sipCode(reason string) int {
	switch reason {
	case "ocs_unavailable", "routing_unavailable", "sbc_unavailable",
		"fraud_unavailable", "no_rate", "internal_error":
		return http.StatusServiceUnavailable
	default:
		return http.StatusForbidden
	}
}

// userPart extracts the user portion of a SIP URI, e.g. "sip:2782111@h" -> "2782111".
func userPart(uri string) string {
	uri = strings.TrimPrefix(uri, "sip:")
	uri = strings.TrimPrefix(uri, "sips:")
	if i := strings.IndexByte(uri, '@'); i >= 0 {
		uri = uri[:i]
	}
	if i := strings.IndexByte(uri, ';'); i >= 0 {
		uri = uri[:i]
	}
	return uri
}
```

```go
	var out struct {
		Stopped bool `json:"stopped"`
	}
	err := f.do(ctx, http.MethodPost, "/stop", map[string]string{"call_id": id}, &out)
	return out.Stopped, err
}

func (f *facadeClient) live(ctx context.Context) (map[string]bool, error) {
	var out struct {
		Calls []struct {
			CallID string `json:"call_id"`
		} `json:"calls"`
	}
	if err := f.do(ctx, http.MethodGet, "/live", nil, &out); err != nil {
		return nil, err
	}
	m := make(map[string]bool, len(out.Calls))
	for _, c := range out.Calls {
		m[c.CallID] = true
	}
	return m, nil
}

func (f *facadeClient) do(ctx context.Context, method, path string, in, out any) error {
	var body io.Reader
	if in != nil {
		b, err := json.Marshal(in)
		if err != nil {
			return err
		}
		body = bytes.NewReader(b)
	}
	req, err := http.NewRequestWithContext(ctx, method, f.base+path, body)
	if err != nil {
		return err
	}
	if in != nil {
		req.Header.Set("Content-Type", "application/json")
	}
	resp, err := f.http.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		b, _ := io.ReadAll(io.LimitReader(resp.Body, 512))
		return fmt.Errorf("%s %s: status %d: %s", method, path, resp.StatusCode, strings.TrimSpace(string(b)))
	}
	return json.NewDecoder(resp.Body).Decode(out)
```

```go
package main

import (
	"io"
	"net/http"
	"strings"
	"sync"

	"go.opentelemetry.io/otel"
	"go.opentelemetry.io/otel/attribute"
	"go.opentelemetry.io/otel/trace"
)

type entry struct {
	switchingID string
	hEntry      string
	hID         string
}

// store maps a Kamailio Call-ID to the live switching call it started.
type store struct {
	mu sync.Mutex
	m  map[string]entry
}

func newStore() *store { return &store{m: map[string]entry{}} }

func (s *store) get(id string) (entry, bool) {
	s.mu.Lock()
	defer s.mu.Unlock()
	e, ok := s.m[id]
	return e, ok
}

func (s *store) put(id string, e entry) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.m[id] = e
}

func (s *store) take(id string) (entry, bool) {
	s.mu.Lock()
	defer s.mu.Unlock()
	e, ok := s.m[id]
	delete(s.m, id)
	return e, ok
}

func (s *store) all() map[string]entry {
	s.mu.Lock()
	defer s.mu.Unlock()
	out := make(map[string]entry, len(s.m))
	for k, v := range s.m {
		out[k] = v
	}
	return out
}

type sipedgeServer struct {
	accounts map[string]string
	facade   *facadeClient
	store    *store
}

func (s *sipedgeServer) invite(w http.ResponseWriter, r *http.Request) {
	body, _ := io.ReadAll(io.LimitReader(r.Body, 4096))
	parts := strings.Split(strings.TrimSpace(string(body)), "|")
	if len(parts) != 7 {
		http.Error(w, "bad invite body", http.StatusBadRequest)
		return
	}
	ci, srcIP, from, to, ruri, hEntry, hID := parts[0], parts[1], parts[2], parts[3], parts[4], parts[5], parts[6]
	ctx, span := otel.Tracer("omnifoni/sipedge").Start(r.Context(), "sipedge.invite",
		trace.WithAttributes(attribute.String("source_ip", srcIP)))
	defer span.End()

	account, ok := s.accounts[srcIP]
	if !ok {
		span.SetAttributes(attribute.String("decision", "ip_blocked"))
		http.Error(w, "ip_blocked", http.StatusForbidden)
		return
	}
	start, err := s.facade.start(ctx, startReq{
		AccountID: account, Destination: userPart(ruri),
		CLI: userPart(from), SourceIP: srcIP,
	})
	if err != nil {
		span.SetAttributes(attribute.String("decision", "policy_unavailable"))
		http.Error(w, "policy_unavailable", http.StatusServiceUnavailable)
		return
	}
	if !start.Allowed {
		span.SetAttributes(attribute.String("decision", start.Reason))
		http.Error(w, start.Reason, sipCode(start.Reason))
		return
	}
	span.SetAttributes(attribute.String("decision", "allowed"),
		attribute.String("call_id", start.CallID))
	s.store.put(ci, entry{switchingID: start.CallID, hEntry: hEntry, hID: hID})
	w.WriteHeader(http.StatusOK)
	_, _ = w.Write([]byte(start.TargetURI))
}

func (s *sipedgeServer) bye(w http.ResponseWriter, r *http.Request) {
	ctx, span := otel.Tracer("omnifoni/sipedge").Start(r.Context(), "sipedge.bye")
	defer span.End()
	body, _ := io.ReadAll(io.LimitReader(r.Body, 1024))
	id := strings.TrimSpace(strings.TrimPrefix(strings.TrimSpace(string(body)), "BYE "))
	if e, ok := s.store.take(id); ok {
		span.SetAttributes(attribute.String("call_id", e.switchingID))
		_, _ = s.facade.stop(ctx, e.switchingID)
	}
	w.WriteHeader(http.StatusOK)
}
```

```go
package main

import (
	"context"
	"log/slog"
	"time"
)

// dialogKiller ends a Kamailio dialog by hash coordinates.
type dialogKiller interface {
	endDialog(ctx context.Context, hEntry, hID string) error
}

type reaper struct {
	store *store
	kam   dialogKiller
	live  func(ctx context.Context) (map[string]bool, error)
}

func newReaper(f *facadeClient, kam *kamRPC, st *store) *reaper {
	return &reaper{store: st, kam: kam, live: f.live}
}

func (r *reaper) run(interval time.Duration) {
	t := time.NewTicker(interval)
	defer t.Stop()
	for range t.C {
		ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
		if err := r.tick(ctx); err != nil {
			slog.Warn("reaper tick", "err", err)
		}
		cancel()
	}
}

// tick ends every tracked dialog whose switching call is no longer live.
func (r *reaper) tick(ctx context.Context) error {
	live, err := r.live(ctx)
	if err != nil {
		return err
	}
	for id, e := range r.store.all() {
		if live[e.switchingID] {
			continue
		}
		if err := r.kam.endDialog(ctx, e.hEntry, e.hID); err != nil {
			slog.Warn("end dialog", "call", id, "err", err)
			continue
		}
		r.store.take(id)
	}
	return nil
}
```

```
		exit;

		exit;
```

```go
// Command sipuas is a scripted SIP user-agent server used by the SIP edge
// smoke and end-to-end tests. It answers INVITE with 200 OK + SDP and BYE
// with 200 OK. It is deliberately minimal: no registration, no re-INVITE.
package main

import (
	"fmt"
	"log/slog"
	"net"
	"os"
	"strings"
	"time"
)

func main() {
	addr := os.Getenv("SIPUAS_ADDR")
	if addr == "" {
		addr = ":5060"
	}
	pc, err := net.ListenPacket("udp", addr)
	if err != nil {
		slog.Error("listen", "err", err)
		os.Exit(1)
	}
	slog.Info("sipuas up", "addr", addr)
	buf := make([]byte, 65535)
	for {
		n, from, err := pc.ReadFrom(buf)
		if err != nil {
			continue
		}
		msg := string(buf[:n])
		switch {
		case strings.HasPrefix(msg, "INVITE"):
			slog.Info("invite", "from", from.String())
			_ = send(pc, from, reply(msg, "200", "OK", sdpBody(from)))
		case strings.HasPrefix(msg, "BYE"):
			slog.Info("bye", "from", from.String())
			_ = send(pc, from, reply(msg, "200", "OK", ""))
		case strings.HasPrefix(msg, "CANCEL"):
			_ = send(pc, from, reply(msg, "200", "OK", ""))
		}
	}
}

func send(pc net.PacketConn, to net.Addr, msg string) error {
	_, err := pc.WriteTo([]byte(msg), to)
	return err
}

func reply(req, code, reason, body string) string {
	lines := strings.Split(req, "\r\n")
	var via, from, to, callID, cseq string
	for _, l := range lines {
		switch {
		case strings.HasPrefix(l, "Via:"):
			via = l
		case strings.HasPrefix(l, "From:"):
			from = l
		case strings.HasPrefix(l, "To:"):
			to = l
		case strings.HasPrefix(l, "Call-ID:"):
			callID = l
		case strings.HasPrefix(l, "CSeq:"):
			cseq = l
		}
	}
	toTag := ";tag=sipuas-" + fmt.Sprint(time.Now().UnixNano())
	if !strings.Contains(to, ";tag=") {
		to += toTag
	}
	var b strings.Builder
	fmt.Fprintf(&b, "SIP/2.0 %s %s\r\n", code, reason)
	b.WriteString(via + "\r\n")
	b.WriteString("From: " + strings.TrimPrefix(from, "From:") + "\r\n")
	b.WriteString("To: " + strings.TrimPrefix(to, "To:") + "\r\n")
	b.WriteString("Call-ID: " + strings.TrimPrefix(callID, "Call-ID:") + "\r\n")
	b.WriteString("CSeq: " + strings.TrimPrefix(cseq, "CSeq:") + "\r\n")
	b.WriteString("Contact: <sip:sipuas@localhost>\r\n")
	b.WriteString("Server: sipuas\r\n")
	if body != "" {
		b.WriteString("Content-Type: application/sdp\r\n")
		fmt.Fprintf(&b, "Content-Length: %d\r\n", len(body))
	}
	b.WriteString("\r\n")
	b.WriteString(body)
	return b.String()
}

func sdpBody(from net.Addr) string {
	host, _, _ := net.SplitHostPort(from.String())
	return "v=0\r\no=sipuas 0 0 IN IP4 " + host + "\r\ns=-\r\nc=IN IP4 " + host +
		"\r\nt=0 0\r\nm=audio 40000 RTP/AVP 0\r\n"
}
```

```go
// Command sipsmoke drives one SIP call through the edge and asserts the
// expected outcome. Modes: answer (default), deny, exhaust.
package main

import (
	"fmt"
	"net"
	"os"
	"strings"
	"time"
)

func main() {
	target := env("SIP_SMOKE_TARGET", "127.0.0.1:5060")
	dest := env("SIP_SMOKE_DEST", "27821110000")
	from := env("SIP_SMOKE_FROM", "27820000000")
	mode := env("SIP_SMOKE_MODE", "answer")

	conn, err := net.Dial("udp", target)
	if err != nil {
		fail("dial: %v", err)
	}
	defer conn.Close()
	_ = conn.SetDeadline(time.Now().Add(8 * time.Second))

	callID := fmt.Sprintf("sipsmoke-%d", time.Now().UnixNano())
	tag := fmt.Sprintf("%d", time.Now().UnixNano())
	via := fmt.Sprintf("Via: SIP/2.0/UDP %s;branch=z9hG4bK%s", localAddr(conn), tag)
	invite := "INVITE sip:" + dest + "@edge SIP/2.0\r\n" +
		via + "\r\n" +
		"Max-Forwards: 70\r\n" +
		"From: <sip:" + from + ">;tag=" + tag + "\r\n" +
		"To: <sip:" + dest + ">\r\n" +
		"Call-ID: " + callID + "\r\n" +
		"CSeq: 1 INVITE\r\n" +
		"Contact: <sip:sipsmoke@" + localAddr(conn) + ">\r\n" +
		"Content-Length: 0\r\n\r\n"

	buf := make([]byte, 65535)
	var final string
	deadline := time.Now().Add(6 * time.Second)
	for time.Now().Before(deadline) {
		if _, err := conn.Write([]byte(invite)); err != nil {
			fail("send invite: %v", err)
		}
		n, err := conn.Read(buf)
		if err != nil {
			if ne, ok := err.(net.Error); ok && ne.Timeout() {
				continue
			}
			fail("read: %v", err)
		}
		final = string(buf[:n])
		if strings.HasPrefix(final, "SIP/2.0 200") || strings.HasPrefix(final, "SIP/2.0 4") || strings.HasPrefix(final, "SIP/2.0 5") || strings.HasPrefix(final, "SIP/2.0 6") {
			break
		}
	}
	if final == "" {
		fail("no final response")
	}

	switch mode {
	case "deny":
		if !strings.HasPrefix(final, "SIP/2.0 403") {
			fail("want 403, got: %s", firstLine(final))
		}
		fmt.Println("ok: sip deny")
		return
	}

	if !strings.HasPrefix(final, "SIP/2.0 200") {
		fail("want 200, got: %s", firstLine(final))
	}
	ack := "ACK sip:" + dest + "@edge SIP/2.0\r\n" + via + "\r\n" +
		"Max-Forwards: 70\r\n" +
		"From: <sip:" + from + ">;tag=" + tag + "\r\n" +
		"To: " + toLine(final) + "\r\n" +
		"Call-ID: " + callID + "\r\n" +
		"CSeq: 1 ACK\r\nContent-Length: 0\r\n\r\n"
	_, _ = conn.Write([]byte(ack))

	if mode == "exhaust" {
		// Wait for the edge to hang up after the prepaid balance runs out.
		_ = conn.SetDeadline(time.Now().Add(4 * time.Second))
		n, err := conn.Read(buf)
		if err != nil {
			fail("no teardown bye within 4s: %v", err)
		}
		if !strings.HasPrefix(string(buf[:n]), "BYE") {
			fail("want BYE, got: %s", firstLine(string(buf[:n])))
		}
		fmt.Println("ok: sip exhaust")
		return
	}

	time.Sleep(1500 * time.Millisecond)
	bye := "BYE sip:" + dest + "@edge SIP/2.0\r\n" + via + "\r\n" +
		"Max-Forwards: 70\r\n" +
		"From: <sip:" + from + ">;tag=" + tag + "\r\n" +
		"To: " + toLine(final) + "\r\n" +
		"Call-ID: " + callID + "\r\n" +
		"CSeq: 2 BYE\r\nContent-Length: 0\r\n\r\n"
	_, _ = conn.Write([]byte(bye))
	_ = conn.SetDeadline(time.Now().Add(3 * time.Second))
	n, err := conn.Read(buf)
	if err != nil || !strings.HasPrefix(string(buf[:n]), "SIP/2.0 200") {
		fail("bye not answered 200: %v", err)
	}
	fmt.Println("ok: sip")
}

func env(key, def string) string {
	if v := os.Getenv(key); v != "" {
		return v
	}
	return def
}

func fail(format string, args ...any) {
	fmt.Fprintf(os.Stderr, format+"\n", args...)
	os.Exit(1)
}

func firstLine(s string) string {
	if i := strings.IndexByte(s, '\n'); i >= 0 {
		s = s[:i]
	}
	return strings.TrimSpace(s)
}

func toLine(resp string) string {
	for _, l := range strings.Split(resp, "\r\n") {
		if strings.HasPrefix(l, "To:") {
			return strings.TrimSpace(strings.TrimPrefix(l, "To:"))
		}
	}
	return ""
}

func localAddr(c net.Conn) string {
	if a, ok := c.LocalAddr().(*net.UDPAddr); ok {
		return fmt.Sprintf("%s:%d", a.IP, a.Port)
	}
	return "127.0.0.1:5060"
}
```

- [ ] **Step 1: Write the e2e script**

Create `ops/sip-e2e.sh`:

```bash
#!/usr/bin/env bash
set -euo pipefail
cd "$(dirname "$0")/.."

fail() { echo "FAIL: $1" >&2; exit 1; }

seed_rate() {
  docker exec omni-fon-postgres-1 psql -U postgres -d catalogue -qAt -c \
    "INSERT INTO rates(prefix,rate_per_sec) VALUES ('$1','$2') ON CONFLICT (prefix) DO UPDATE SET rate_per_sec=EXCLUDED.rate_per_sec" >/dev/null
}
seed_account() {
  docker exec omni-fon-postgres-1 psql -U postgres -d charging -qAt -c \
    "INSERT INTO accounts(account_id,kind,balance,credit_limit) VALUES ('$1','prepaid',$2,0) ON CONFLICT (account_id) DO UPDATE SET balance=EXCLUDED.balance" >/dev/null
}
seed_route() {
  docker exec omni-fon-postgres-1 psql -U postgres -d routing -qAt -c \
    "INSERT INTO trunks(trunk_id,enabled,target_uri) VALUES ('$1',true,'sip:sipuas:5060') ON CONFLICT (trunk_id) DO UPDATE SET enabled=true,target_uri=EXCLUDED.target_uri" >/dev/null
  docker exec omni-fon-postgres-1 psql -U postgres -d routing -qAt -c \
    "INSERT INTO lcr_routes(prefix,trunk_id,cost_per_sec) VALUES ('$2','$1',1) ON CONFLICT (prefix,trunk_id) DO UPDATE SET cost_per_sec=EXCLUDED.cost_per_sec" >/dev/null
}

docker compose up -d sipuas

# SIPEDGE_TRUNK_ACCOUNTS maps the one smoke source IP (172.28.0.1) to one
# account, so all three scenarios run as acc-sip-smoke.

# 1. Golden answer path + CDR.
seed_rate 2782 0.010000
seed_account acc-sip-smoke 100000
seed_route trunk-e2e 2782
(cd services/sipedge && SIP_SMOKE_DEST=27821110000 go run ./cmd/sipsmoke) || fail "answer call"
cdr=$(docker exec omni-fon-kafka-1 /opt/kafka/bin/kafka-console-consumer.sh \
  --bootstrap-server localhost:9092 --topic cdrs.normalized \
  --from-beginning --max-messages 50 --timeout-ms 5000 2>/dev/null | grep -c 'acc-sip-smoke' || true)
[ "$cdr" -ge 1 ] || fail "no cdr for acc-sip-smoke"
echo "ok: sip e2e answer + cdr"

# 2. Denial path: barring rule returns 403.
docker exec omni-fon-postgres-1 psql -U postgres -d sbc -qAt -c \
  "INSERT INTO barring_rules(match_type,prefix,action) SELECT 'destination','27829999','deny' WHERE NOT EXISTS (SELECT 1 FROM barring_rules WHERE prefix='27829999')" >/dev/null
seed_rate 2783 0.010000
seed_account acc-sip-smoke 100000
seed_route trunk-deny 2783
(cd services/sipedge && SIP_SMOKE_DEST=27829990000 SIP_SMOKE_MODE=deny go run ./cmd/sipsmoke) || fail "deny call"
echo "ok: sip e2e deny"

# 3. Exhaustion path: 1 s of authorization at a high rate, expect BYE
# within 4 s (balance 100000 minor / rate 100000 minor per second = 1 s).
seed_rate 2788 100000.000000
seed_account acc-sip-smoke 100000
seed_route trunk-exh 2788
(cd services/sipedge && SIP_SMOKE_DEST=27881110000 SIP_SMOKE_MODE=exhaust go run ./cmd/sipsmoke) || fail "exhaust call"
echo "ok: sip e2e exhaust"

echo "ok: sip e2e"
```
