# Reports Slice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `services/reports`: ingest `cdrs.normalized` into ClickHouse with billing-parity rating, and serve CDR-list and grouped-summary reports over gRPC and an authenticated HTTP facade.

**Architecture:** One Go binary owns a Kafka consumer, the `reports` ClickHouse database and both API surfaces (payments precedent). Each CDR is rated at ingestion through Catalogue `GetRate` using the OCS `ROUND(seconds*rate)` rule; rows are deduplicated by `ReplacingMergeTree` keyed on `(account_id, started_at, call_id)`. The HTTP facade authenticates against identity and scopes account IDs to the caller's tenant before invoking the same report methods as gRPC.

**Tech Stack:** Go 1.27.4, `github.com/ClickHouse/clickhouse-go/v2` v2.51.0, franz-go v1.22.3, gRPC v1.85.0, protobuf v1.36.12, `omnifoni/observability`, ClickHouse 24.10 (compose).

## Global Constraints

- Clean-room: never read ASTPP source code, schemas, or licensed assets.
- FOSS-only stack; no Terraform, no Redis.
- Money is integer minor units; rate is `numeric(14,6)` minor units per second; per-event cost is `ROUND(seconds*rate)` half away from zero.
- Stubs are generated only by `make protoc`; never hand-edit `gen/`.
- Repo modules live in `services/<name>`; sibling imports use `replace omnifoni/<sibling> => ../<sibling>`.
- Tests hit real dev infrastructure: Postgres/Kafka/ClickHouse/Catalogue on localhost; Kafka advertises `kafka:9092`, so `/etc/hosts` must map `127.0.0.1 kafka`.
- After code changes rebuild the image before `make up`: `docker compose build <svc> && docker compose up -d <svc>`.
- `make test` = `go vet ./... && go test ./...` per module, no `-race`; verify with `make test 2>&1 | grep -E "^FAIL"` (empty = pass).
- Request-scoped logs use `slog.*Context`; spans use `otel.Tracer(name)` inside the function, never a package-level cached tracer.
- Telemetry is fail-open; never fail a request because the exporter is down.
- Defaults: ClickHouse DSN `clickhouse://omnifoni:dev-only@clickhouse:9000`; gRPC `:50061`; HTTP `:50062`.

`services/reports/money.go`:

```go
package main

import (
	"fmt"
	"math/big"
)

// RateCost mirrors the OCS money rule: ROUND(seconds * rate_per_sec) in
// integer minor units, half away from zero.
func RateCost(seconds int64, rate string) (int64, error) {
	r, ok := new(big.Rat).SetString(rate)
	if !ok {
		return 0, fmt.Errorf("invalid rate %q", rate)
	}
	r.Mul(r, new(big.Rat).SetInt64(seconds))

	num, den := r.Num(), r.Denom() // den > 0
	q, rem := new(big.Int).QuoRem(num, den, new(big.Int))
	rem.Abs(rem)
	rem.Lsh(rem, 1) // 2*|rem| >= den means a half or more
	if rem.Cmp(den) >= 0 {
		if r.Sign() < 0 {
			q.Sub(q, big.NewInt(1))
		} else {
			q.Add(q, big.NewInt(1))
		}
	}
	if !q.IsInt64() {
		return 0, fmt.Errorf("rate cost overflow for %d * %s", seconds, rate)
	}
	return q.Int64(), nil
}
```

```go
package main

import (
	"context"
	"fmt"
	"strings"
	"time"

	"github.com/ClickHouse/clickhouse-go/v2"
)

type CdrRow struct {
	CallID, AccountID, Destination, CLI string
	StartedAt, EndedAt                  time.Time
	ChargedSeconds                      int64
	EndReason, TrunkID                  string
	RatePerSec                          string
	AmountMinor                         int64
	RateMissing                         bool
}

type Filter struct {
	AccountIDs        []string
	From, To          time.Time
	DestinationPrefix string
	TrunkID, EndReason string
}

type Cursor struct {
	StartedAt time.Time
	CallID    string
}

type GroupBy string

const (
	GroupDay         GroupBy = "day"
	GroupDestination GroupBy = "destination"
	GroupAccount     GroupBy = "account"
	GroupTrunk       GroupBy = "trunk"
)

type SummaryRow struct {
	Key         string
	Calls       int64
	Seconds     int64
	AmountMinor int64
}

type Store struct {
	db clickhouse.Conn
}

const schemaDDL = `CREATE DATABASE IF NOT EXISTS reports`

const tableDDL = `
CREATE TABLE IF NOT EXISTS reports.cdrs (
  call_id         String,
  account_id      String,
  destination     String,
  cli             String,
  started_at      DateTime64(3, 'UTC'),
  ended_at        DateTime64(3, 'UTC'),
  charged_seconds Int64,
  end_reason      LowCardinality(String),
  trunk_id        LowCardinality(String),
  rate_per_sec    Decimal(14, 6),
  amount_minor    Int64,
  rate_missing    Bool,
  ingested_at     DateTime64(3, 'UTC') DEFAULT now64(3)
) ENGINE = ReplacingMergeTree(ingested_at)
PARTITION BY toYYYYMM(started_at)
ORDER BY (account_id, started_at, call_id)`

func Open(ctx context.Context, dsn string) (*Store, error) {
	opts, err := clickhouse.ParseDSN(dsn)
	if err != nil {
		return nil, err
	}
	db, err := clickhouse.Open(opts)
	if err != nil {
		return nil, err
	}
	if err := db.Ping(ctx); err != nil {
		_ = db.Close()
		return nil, err
	}
	return &Store{db: db}, nil
}

func (s *Store) Close() error { return s.db.Close() }

func (s *Store) Migrate(ctx context.Context) error {
	if err := s.db.Exec(ctx, schemaDDL); err != nil {
		return err
	}
	return s.db.Exec(ctx, tableDDL)
}

func (s *Store) Insert(ctx context.Context, rows []CdrRow) error {
	if len(rows) == 0 {
		return nil
	}
	b, err := s.db.PrepareBatch(ctx, `INSERT INTO reports.cdrs
		(call_id, account_id, destination, cli, started_at, ended_at, charged_seconds,
		 end_reason, trunk_id, rate_per_sec, amount_minor, rate_missing)`)
	if err != nil {
		return err
	}
	for _, r := range rows {
		if err := b.Append(r.CallID, r.AccountID, r.Destination, r.CLI, r.StartedAt, r.EndedAt,
			r.ChargedSeconds, r.EndReason, r.TrunkID, r.RatePerSec, r.AmountMinor, r.RateMissing); err != nil {
			_ = b.Abort()
			return err
		}
	}
	return b.Send()
}

func (f Filter) where() (string, []any) {
	clauses := []string{
		"started_at >= toDateTime64(?, 3, 'UTC')",
		"started_at < toDateTime64(?, 3, 'UTC')",
	}
	args := []any{f.From.UTC(), f.To.UTC()}
	if len(f.AccountIDs) > 0 {
		clauses = append(clauses, "account_id IN (?)")
		args = append(args, f.AccountIDs)
	}
	if f.DestinationPrefix != "" {
		clauses = append(clauses, "startsWith(destination, ?)")
		args = append(args, f.DestinationPrefix)
	}
	if f.TrunkID != "" {
		clauses = append(clauses, "trunk_id = ?")
		args = append(args, f.TrunkID)
	}
	if f.EndReason != "" {
		clauses = append(clauses, "end_reason = ?")
		args = append(args, f.EndReason)
	}
	return strings.Join(clauses, " AND "), args
}

func (s *Store) ListCdrs(ctx context.Context, f Filter, limit int, cur *Cursor) ([]CdrRow, error) {
	where, args := f.where()
	if cur != nil {
		where += " AND (started_at, call_id) < (toDateTime64(?, 3, 'UTC'), ?)"
		args = append(args, cur.StartedAt.UTC(), cur.CallID)
	}
	q := `SELECT call_id, account_id, destination, cli,
			toUnixTimestamp64Milli(started_at), toUnixTimestamp64Milli(ended_at),
			charged_seconds, end_reason, trunk_id, toString(rate_per_sec),
			amount_minor, rate_missing
		FROM reports.cdrs FINAL
		WHERE ` + where + `
		ORDER BY started_at DESC, call_id DESC
		LIMIT ?`
	args = append(args, limit)

	rs, err := s.db.Query(ctx, q, args...)
	if err != nil {
		return nil, err
	}
	defer rs.Close()

	var out []CdrRow
	for rs.Next() {
		var r CdrRow
		var startMS, endMS int64
		if err := rs.Scan(&r.CallID, &r.AccountID, &r.Destination, &r.CLI,
			&startMS, &endMS, &r.ChargedSeconds, &r.EndReason, &r.TrunkID,
			&r.RatePerSec, &r.AmountMinor, &r.RateMissing); err != nil {
			return nil, err
		}
		r.StartedAt = time.UnixMilli(startMS).UTC()
		r.EndedAt = time.UnixMilli(endMS).UTC()
		out = append(out, r)
	}
	return out, rs.Err()
}

func keyExpr(g GroupBy) (string, error) {
	switch g {
	case GroupDay:
		return "toString(toDate(started_at))", nil
	case GroupDestination:
		return "destination", nil
	case GroupAccount:
		return "account_id", nil
	case GroupTrunk:
		return "trunk_id", nil
	}
	return "", fmt.Errorf("unknown group_by %q", g)
}

func (s *Store) Summarize(ctx context.Context, f Filter, g GroupBy) ([]SummaryRow, error) {
	key, err := keyExpr(g)
	if err != nil {
		return nil, err
	}
	where, args := f.where()
	q := `SELECT ` + key + ` AS k, count(), sum(charged_seconds), sum(amount_minor)
		FROM reports.cdrs FINAL
		WHERE ` + where + `
		GROUP BY k
		ORDER BY sum(amount_minor) DESC, k ASC`

	rs, err := s.db.Query(ctx, q, args...)
	if err != nil {
		return nil, err
	}
	defer rs.Close()

	var out []SummaryRow
	for rs.Next() {
		var row SummaryRow
		var calls uint64
```

```go
// failures return an error so the consumer retries; they never zero money.
type Ingester struct {
	store *Store
	cat   cataloguepb.CatalogueClient
}

func NewIngester(store *Store, cat cataloguepb.CatalogueClient) *Ingester {
	return &Ingester{store: store, cat: cat}
}

// Validate mirrors billing's CDR requirements.
func Validate(c *mediationpb.CallCdr) string {
	switch {
	case c.GetCallId() == "":
		return "missing_call_id"
	case c.GetAccountId() == "":
		return "missing_account_id"
	case c.GetStartedAtMs() == 0:
		return "missing_started_at"
	case c.GetChargedSeconds() < 0:
		return "negative_charged_seconds"
	case c.GetEndedAtMs() < c.GetStartedAtMs():
		return "ended_before_started"
	}
	return ""
}

func (i *Ingester) Handle(ctx context.Context, rec *kgo.Record) error {
	c := &mediationpb.CallCdr{}
	if err := proto.Unmarshal(rec.Value, c); err != nil {
		slog.ErrorContext(ctx, "cdr rejected", "reason", "decode_error", "err", err)
		return nil // poison pill: log and move on
	}
	ctx = observability.ExtractKafka(ctx, rec)
	ctx, span := observability.ConsumerSpan(ctx, "reports.ingest", c.GetCallId())
	defer span.End()

	if reason := Validate(c); reason != "" {
		slog.ErrorContext(ctx, "cdr rejected", "reason", reason)
		return nil
	}
	row, err := i.rate(ctx, c)
	if err != nil {
		return err
	}
	return i.store.Insert(ctx, []CdrRow{row})
}

func (i *Ingester) rate(ctx context.Context, c *mediationpb.CallCdr) (CdrRow, error) {
	start := time.UnixMilli(c.GetStartedAtMs()).UTC()
	row := CdrRow{
		CallID:         c.GetCallId(),
		AccountID:      c.GetAccountId(),
		Destination:    c.GetDestination(),
		CLI:            c.GetCli(),
		StartedAt:      start,
		EndedAt:        time.UnixMilli(c.GetEndedAtMs()).UTC(),
		ChargedSeconds: c.GetChargedSeconds(),
		EndReason:      c.GetEndReason(),
		TrunkID:        c.GetTrunkId(),
	}
	resp, err := i.cat.GetRate(ctx, &cataloguepb.GetRateRequest{Destination: c.GetDestination()})
	if err != nil {
		return CdrRow{}, fmt.Errorf("catalogue get rate: %w", err)
	}
	if !resp.GetFound() {
		row.RatePerSec = "0"
		row.RateMissing = true
		return row, nil
	}
	amount, err := RateCost(c.GetChargedSeconds(), resp.GetRatePerSec())
	if err != nil {
		return CdrRow{}, err
	}
	row.RatePerSec = resp.GetRatePerSec()
	row.AmountMinor = amount
	return row, nil
}
```

```go
// Run consumes CDRs in small batches, handling records sequentially. A failed
// record is retried in-process with backoff; offsets are committed only after
// every record of the batch was handled. Keys pin a call to one partition, so
// this is serial per call.
package main

import (
	"context"
	"log/slog"
	"strconv"
	"time"

	kgo "github.com/twmb/franz-go/pkg/kgo"
)

type Handler interface {
	Handle(ctx context.Context, rec *kgo.Record) error
}

func Run(ctx context.Context, cl *kgo.Client, h Handler) {
	for {
		fetches := cl.PollRecords(ctx, 100)
		fetches.EachError(func(topic string, partition int32, err error) {
			slog.ErrorContext(ctx, "kafka fetch", "topic", topic, "partition", partition, "err", err)
		})
		lasts := map[string]*kgo.Record{}
		for iter := fetches.RecordIter(); !iter.Done(); {
			rec := iter.Next()
			for {
				err := h.Handle(ctx, rec)
				if err == nil {
					break
				}
				if ctx.Err() != nil {
					return
				}
				slog.ErrorContext(ctx, "handle failed, retrying", "err", err)
				select {
				case <-ctx.Done():
					return
				case <-time.After(time.Second):
				}
			}
			lasts[rec.Topic+"/"+strconv.Itoa(int(rec.Partition))] = rec
		}
		if ctx.Err() != nil {
			return
		}
		var toCommit []*kgo.Record
		for _, rec := range lasts {
			toCommit = append(toCommit, rec)
		}
		if len(toCommit) > 0 {
			if err := cl.CommitRecords(ctx, toCommit...); err != nil && ctx.Err() == nil {
				slog.ErrorContext(ctx, "commit failed", "err", err)
			}
		}
	}
}
```

```go
package main

import (
	"context"
	"encoding/base64"
	"errors"
	"fmt"
	"strconv"
	"strings"
	"time"

	"go.opentelemetry.io/otel"
	"go.opentelemetry.io/otel/attribute"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/status"

	pb "omnifoni/reports/gen/reportspb"
)

type reportsServer struct {
	store *Store
}

const (
	defaultLimit = 100
	maxLimit     = 1000
)

func parseBound(s string, endOfDay bool) (time.Time, error) {
	if t, err := time.Parse(time.RFC3339, s); err == nil {
		return t.UTC(), nil
	}
	t, err := time.Parse("2006-01-02", s)
	if err != nil {
		return time.Time{}, fmt.Errorf("invalid time %q", s)
	}
	if endOfDay {
		return t.Add(24 * time.Hour), nil
	}
	return t, nil
}

func parseFilter(f *pb.CdrFilter) (Filter, error) {
	if f.GetFrom() == "" || f.GetTo() == "" {
		return Filter{}, errors.New("from and to are required")
	}
	from, err := parseBound(f.GetFrom(), false)
	if err != nil {
		return Filter{}, err
	}
	to, err := parseBound(f.GetTo(), true)
	if err != nil {
		return Filter{}, err
	}
	if !to.After(from) {
		return Filter{}, errors.New("to must be after from")
	}
	return Filter{
		AccountIDs:        f.GetAccountIds(),
		From:              from,
		To:                to,
		DestinationPrefix: f.GetDestinationPrefix(),
		TrunkID:           f.GetTrunkId(),
		EndReason:         f.GetEndReason(),
	}, nil
}

func encodeCursor(t time.Time, callID string) string {
	raw := strconv.FormatInt(t.UnixMilli(), 10) + "|" + callID
	return base64.RawURLEncoding.EncodeToString([]byte(raw))
}

func decodeCursor(s string) (*Cursor, error) {
	b, err := base64.RawURLEncoding.DecodeString(s)
	if err != nil {
		return nil, err
	}
	parts := strings.SplitN(string(b), "|", 2)
	if len(parts) != 2 {
		return nil, errors.New("malformed cursor")
	}
	ms, err := strconv.ParseInt(parts[0], 10, 64)
	if err != nil {
		return nil, err
	}
	return &Cursor{StartedAt: time.UnixMilli(ms).UTC(), CallID: parts[1]}, nil
}

func (s *reportsServer) ListCdrs(ctx context.Context, req *pb.ListCdrsRequest) (*pb.ListCdrsResponse, error) {
	f, err := parseFilter(req.GetFilter())
	if err != nil {
		return nil, status.Error(codes.InvalidArgument, err.Error())
	}
	limit := int(req.GetLimit())
	if limit <= 0 {
		limit = defaultLimit
	}
	if limit > maxLimit {
		limit = maxLimit
	}
	var cur *Cursor
	if c := req.GetCursor(); c != "" {
		cur, err = decodeCursor(c)
		if err != nil {
			return nil, status.Error(codes.InvalidArgument, "invalid cursor")
		}
	}

	ctx, span := otel.Tracer("reports").Start(ctx, "reports.list_cdrs")
	span.SetAttributes(attribute.String("from", f.From.Format(time.RFC3339)),
		attribute.String("to", f.To.Format(time.RFC3339)))
	defer span.End()

	rows, err := s.store.ListCdrs(ctx, f, limit+1, cur)
	if err != nil {
		return nil, status.Error(codes.Unavailable, "store unavailable")
	}
	resp := &pb.ListCdrsResponse{Cdrs: []*pb.Cdr{}}
	hasMore := len(rows) > limit
	if hasMore {
		rows = rows[:limit]
	}
	for _, r := range rows {
		resp.Cdrs = append(resp.Cdrs, &pb.Cdr{
			CallId:         r.CallID,
			AccountId:      r.AccountID,
			Destination:    r.Destination,
			Cli:            r.CLI,
			StartedAtMs:    r.StartedAt.UnixMilli(),
			EndedAtMs:      r.EndedAt.UnixMilli(),
			ChargedSeconds: r.ChargedSeconds,
			EndReason:      r.EndReason,
			TrunkId:        r.TrunkID,
			RatePerSec:     r.RatePerSec,
			AmountMinor:    r.AmountMinor,
			RateMissing:    r.RateMissing,
		})
	}
	if hasMore && len(rows) > 0 {
		last := rows[len(rows)-1]
		resp.NextCursor = encodeCursor(last.StartedAt, last.CallID)
	}
	return resp, nil
}

func parseGroup(g pb.GroupBy) (GroupBy, error) {
	switch g {
	case pb.GroupBy_DAY:
		return GroupDay, nil
	case pb.GroupBy_DESTINATION:
		return GroupDestination, nil
	case pb.GroupBy_ACCOUNT:
		return GroupAccount, nil
	case pb.GroupBy_TRUNK:
		return GroupTrunk, nil
	}
	return "", errors.New("group_by is required")
}

func (s *reportsServer) Summarize(ctx context.Context, req *pb.SummarizeRequest) (*pb.SummarizeResponse, error) {
	f, err := parseFilter(req.GetFilter())
	if err != nil {
		return nil, status.Error(codes.InvalidArgument, err.Error())
	}
	group, err := parseGroup(req.GetGroupBy())
	if err != nil {
		return nil, status.Error(codes.InvalidArgument, err.Error())
	}

	ctx, span := otel.Tracer("reports").Start(ctx, "reports.summarize")
	span.SetAttributes(attribute.String("group_by", string(group)),
		attribute.String("from", f.From.Format(time.RFC3339)),
		attribute.String("to", f.To.Format(time.RFC3339)))
	defer span.End()

	rows, err := s.store.Summarize(ctx, f, group)
	if err != nil {
		return nil, status.Error(codes.Unavailable, "store unavailable")
	}
	resp := &pb.SummarizeResponse{Rows: []*pb.SummaryRow{}, Totals: &pb.SummaryRow{}}
	for _, r := range rows {
		resp.Rows = append(resp.Rows, &pb.SummaryRow{
			Key: r.Key, Calls: r.Calls, Seconds: r.Seconds, AmountMinor: r.AmountMinor})
		resp.Totals.Calls += r.Calls
		resp.Totals.Seconds += r.Seconds
		resp.Totals.AmountMinor += r.AmountMinor
	}
	span.SetAttributes(attribute.Int("rows", len(resp.Rows)))
	return resp, nil
}
```

```go
package main

import (
	"context"
	"encoding/json"
	"errors"
	"net/http"
	"net/url"

	"google.golang.org/grpc/status"

// API is the report surface the facade serves; *reportsServer implements it.
type API interface {
	ListCdrs(context.Context, *pb.ListCdrsRequest) (*pb.ListCdrsResponse, error)
	Summarize(context.Context, *pb.SummarizeRequest) (*pb.SummarizeResponse, error)
}

type Authenticator struct {
	base   string
	client *http.Client
}

func NewAuthenticator(base string) *Authenticator {
	return &Authenticator{base: base, client: &http.Client{}}
}

type meResponse struct {
	Role     string `json:"role"`
	TenantID string `json:"tenant_id"`
}

type accountResponse struct {
	AccountID string `json:"account_id"`
}

// Accounts validates the bearer token and returns the account IDs the caller
// may see. Status is the HTTP status to send when err is non-nil.
func (a *Authenticator) Accounts(ctx context.Context, token, requestedTenant string) ([]string, int, error) {
	me, code, err := a.getMe(ctx, token)
	if err != nil {
		return nil, code, err
	}
	if me.Role == "user" {
		return nil, http.StatusForbidden, errors.New("forbidden")
	}
	scope := me.TenantID
	if me.Role == "admin" && requestedTenant != "" {
		scope = requestedTenant
	}
	if scope == "" {
		return nil, http.StatusBadRequest, errors.New("tenant_id required")
	}

	req, err := http.NewRequestWithContext(ctx, http.MethodGet,
		a.base+"/accounts?tenant_id="+url.QueryEscape(scope), nil)
	if err != nil {
		return nil, http.StatusBadGateway, err
	}
	req.Header.Set("Authorization", "Bearer "+token)
	resp, err := a.client.Do(req)
	if err != nil {
		return nil, http.StatusBadGateway, err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return nil, http.StatusBadGateway, errors.New("identity accounts failed")
	}
	var accts []accountResponse
	if err := json.NewDecoder(resp.Body).Decode(&accts); err != nil {
		return nil, http.StatusBadGateway, err
	}
	ids := make([]string, 0, len(accts))
	for _, a := range accts {
		ids = append(ids, a.AccountID)
	}
	return ids, http.StatusOK, nil
}

func (a *Authenticator) getMe(ctx context.Context, token string) (meResponse, int, error) {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, a.base+"/me", nil)
	if err != nil {
		return meResponse{}, http.StatusBadGateway, err
	}
	req.Header.Set("Authorization", "Bearer "+token)
	resp, err := a.client.Do(req)
	if err != nil {
		return meResponse{}, http.StatusBadGateway, err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return meResponse{}, http.StatusUnauthorized, errors.New("unauthorized")
	}
	var me meResponse
	if err := json.NewDecoder(resp.Body).Decode(&me); err != nil {
		return meResponse{}, http.StatusBadGateway, err
	}
	return me, http.StatusOK, nil
}

type httpServer struct {
	api  API
	auth *Authenticator
}

func bearer(r *http.Request) (string, bool) {
	const prefix = "Bearer "
	h := r.Header.Get("Authorization")
	if len(h) <= len(prefix) || h[:len(prefix)] != prefix {
		return "", false
	}
	return h[len(prefix):], true
}

func writeJSON(w http.ResponseWriter, code int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(code)
	_ = json.NewEncoder(w).Encode(v)
}

func writeErr(w http.ResponseWriter, code int, msg string) {
	writeJSON(w, code, map[string]string{"error": msg})
}

func intersect(allowed, requested []string) []string {
	ok := make(map[string]bool, len(allowed))
	for _, id := range allowed {
		ok[id] = true
	}
	out := make([]string, 0, len(requested))
	for _, id := range requested {
		if ok[id] {
			out = append(out, id)
		}
	}
	return out
}

type cdrJSON struct {
	CallID         string `json:"call_id"`
	AccountID      string `json:"account_id"`
	Destination    string `json:"destination"`
	CLI            string `json:"cli"`
	StartedAtMS    int64  `json:"started_at_ms"`
	EndedAtMS      int64  `json:"ended_at_ms"`
	ChargedSeconds int64  `json:"charged_seconds"`
	EndReason      string `json:"end_reason"`
	TrunkID        string `json:"trunk_id"`
	RatePerSec     string `json:"rate_per_sec"`
	AmountMinor    int64  `json:"amount_minor"`
	RateMissing    bool   `json:"rate_missing"`
}

type listJSON struct {
	Cdrs       []cdrJSON `json:"cdrs"`
	NextCursor string    `json:"next_cursor"`
}

type rowJSON struct {
	Key         string `json:"key"`
	Calls       int64  `json:"calls"`
	Seconds     int64  `json:"seconds"`
	AmountMinor int64  `json:"amount_minor"`
}

type summaryJSON struct {
	Rows   []rowJSON `json:"rows"`
	Totals rowJSON   `json:"totals"`
}

func (h *httpServer) ListCdrs(w http.ResponseWriter, r *http.Request) {
	token, ok := bearer(r)
	if !ok {
		writeErr(w, http.StatusUnauthorized, "unauthorized")
		return
	}
	q := r.URL.Query()
	ids, code, err := h.auth.Accounts(r.Context(), token, q.Get("tenant_id"))
	if err != nil {
		writeErr(w, code, err.Error())
		return
	}
	req := &pb.ListCdrsRequest{
		Filter: &pb.CdrFilter{
			AccountIds:        ids,
			From:              q.Get("from"),
			To:                q.Get("to"),
			DestinationPrefix: q.Get("destination_prefix"),
			TrunkId:           q.Get("trunk_id"),
			EndReason:         q.Get("end_reason"),
		},
		Cursor: q.Get("cursor"),
	}
	if v := q.Get("limit"); v != "" {
		n, err := strconv.Atoi(v)
		if err != nil {
			writeErr(w, http.StatusBadRequest, "invalid limit")
			return
		}
		req.Limit = int32(n)
	}
	if requested := q["account_id"]; len(requested) > 0 {
		req.Filter.AccountIds = intersect(ids, requested)
		if len(req.Filter.AccountIds) == 0 {
			writeJSON(w, http.StatusOK, listJSON{Cdrs: []cdrJSON{}})
			return
		}
	}

	resp, err := h.api.ListCdrs(r.Context(), req)
	if err != nil {
		writeErr(w, grpcStatus(err), err.Error())
		return
	}
	out := listJSON{Cdrs: []cdrJSON{}, NextCursor: resp.GetNextCursor()}
	for _, c := range resp.GetCdrs() {
		out.Cdrs = append(out.Cdrs, cdrJSON{
			CallID: c.GetCallId(), AccountID: c.GetAccountId(), Destination: c.GetDestination(),
			CLI: c.GetCli(), StartedAtMS: c.GetStartedAtMs(), EndedAtMS: c.GetEndedAtMs(),
			ChargedSeconds: c.GetChargedSeconds(), EndReason: c.GetEndReason(), TrunkID: c.GetTrunkId(),
			RatePerSec: c.GetRatePerSec(), AmountMinor: c.GetAmountMinor(), RateMissing: c.GetRateMissing(),
		})
	}
	writeJSON(w, http.StatusOK, out)
}

func (h *httpServer) Summarize(w http.ResponseWriter, r *http.Request) {
	token, ok := bearer(r)
	if !ok {
		writeErr(w, http.StatusUnauthorized, "unauthorized")
		return
	}
	q := r.URL.Query()
	ids, code, err := h.auth.Accounts(r.Context(), token, q.Get("tenant_id"))
	if err != nil {
		writeErr(w, code, err.Error())
		return
	}

	if !ok {
		writeErr(w, http.StatusBadRequest, "invalid group_by")
		return

			AccountIds:        ids,
			From:              q.Get("from"),
			To:                q.Get("to"),
			DestinationPrefix: q.Get("destination_prefix"),
			TrunkId:           q.Get("trunk_id"),
			EndReason:         q.Get("end_reason"),
		},
		GroupBy: group,
	}
	if requested := q["account_id"]; len(requested) > 0 {
		req.Filter.AccountIds = intersect(ids, requested)
		if len(req.Filter.AccountIds) == 0 {
			writeJSON(w, http.StatusOK, summaryJSON{Rows: []rowJSON{}, Totals: rowJSON{}})
			return
		}
	}

	resp, err := h.api.Summarize(r.Context(), req)
	if err != nil {
		writeErr(w, grpcStatus(err), err.Error())
		return
	}
	out := summaryJSON{Rows: []rowJSON{}, Totals: rowJSON{
		Key: resp.GetTotals().GetKey(), Calls: resp.GetTotals().GetCalls(),
		Seconds: resp.GetTotals().GetSeconds(), AmountMinor: resp.GetTotals().GetAmountMinor(),
	}}
	for _, row := range resp.GetRows() {
		out.Rows = append(out.Rows, rowJSON{
			Key: row.GetKey(), Calls: row.GetCalls(), Seconds: row.GetSeconds(), AmountMinor: row.GetAmountMinor()})
	}
	writeJSON(w, http.StatusOK, out)
}

func groupFromString(s string) (pb.GroupBy, bool) {
	switch s {
	case "day":
		return pb.GroupBy_DAY, true
	case "destination":
		return pb.GroupBy_DESTINATION, true
	case "account":
		return pb.GroupBy_ACCOUNT, true
	case "trunk":
		return pb.GroupBy_TRUNK, true
	}
	return pb.GroupBy_GROUP_BY_UNSPECIFIED, false
}

func grpcStatus(err error) int {
	if status.Code(err) == codes.InvalidArgument {
		return http.StatusBadRequest
	}
	return http.StatusServiceUnavailable
}
```

```go
	slog.Info("reports gRPC up", "addr", addr, "brokers", brokers, "topic", topic, "group", group)

	mux := http.NewServeMux()
	h := &httpServer{api: api, auth: NewAuthenticator(identity)}
	mux.HandleFunc("GET /reports/cdrs", h.ListCdrs)
	mux.HandleFunc("GET /reports/summary", h.Summarize)
	mux.HandleFunc("GET /healthz", func(w http.ResponseWriter, _ *http.Request) {
		w.WriteHeader(http.StatusOK)
	})

	srv := &http.Server{Addr: httpAddr, Handler: observability.Middleware(mux), ReadHeaderTimeout: 5 * time.Second}
	go func() {
		<-ctx.Done()
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		_ = srv.Shutdown(shutdownCtx)
	}()
	slog.Info("reports HTTP up", "addr", httpAddr, "identity", identity)
	if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
		slog.Error("http serve failed", "err", err)
		os.Exit(1)
	}
}
```

```bash
#!/usr/bin/env bash
# Inserts N synthetic CDRs, times 20 account+window summary queries and prints
# p95 in milliseconds. Not part of `make test`; target is p95 < 2000 ms at N=10M.
set -euo pipefail
cd "$(dirname "$0")/.."

N=${1:-10000000}
ch() {
  docker exec omni-fon-clickhouse-1 clickhouse-client --user omnifoni --password dev-only "$@"
}

echo "inserting $N rows..."
ch --query "
INSERT INTO reports.cdrs
  (call_id, account_id, destination, cli, started_at, ended_at, charged_seconds,
   end_reason, trunk_id, rate_per_sec, amount_minor, rate_missing)
SELECT concat('bench-', toString(number)),
       concat('acct-', toString(number % 100)),
       concat('2782', toString(number % 10000)),
       '2721000',
       toDateTime64('2026-01-01 00:00:00', 3, 'UTC') + toIntervalSecond(number % 7776000),
       toDateTime64('2026-01-01 00:00:00', 3, 'UTC') + toIntervalSecond(number % 7776000) + 60,
       60, 'normal', concat('trunk-', toString(number % 5)),
       toDecimal64('0.012345', 6), 1, false
FROM numbers($N)"

times=()
for _ in $(seq 1 20); do
  t=$(ch --time --format Null --query "
    SELECT count(), sum(amount_minor)
    FROM reports.cdrs FINAL
    WHERE account_id IN ('acct-1','acct-2')
      AND started_at >= toDateTime64('2026-01-01 00:00:00', 3, 'UTC')
      AND started_at <  toDateTime64('2026-03-01 00:00:00', 3, 'UTC')" 2>&1 | tail -1)
  times+=("$t")
done

p95=$(printf '%s\n' "${times[@]}" | sort -n | awk '{a[NR]=$1} END {print a[int(NR*0.95)]}')
p95_ms=$(awk -v t="$p95" 'BEGIN {printf "%.0f", t*1000}')
echo "p95: ${p95_ms} ms"
awk -v ms="$p95_ms" 'BEGIN {if (ms > 2000) {print "WARN: p95 above the 2000 ms target"; exit 1}}'
echo "bench: PASS"

ch --query "DELETE FROM reports.cdrs WHERE call_id LIKE 'bench-%'"
echo "cleaned bench rows"
```
