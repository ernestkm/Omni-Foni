# Toll-fraud MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the P6 toll-fraud MVP as `services/fraud` with hard fraud rules, versioned advisory scoring/alerts, normalized-CDR ingestion, SBC pre-trunk enforcement, and a replay baseline.

**Architecture:** A new Fraud gRPC service owns fraud controls/usage/alerts in Postgres and consumes `cdrs.normalized` from Kafka to convert call reservations into completed usage. SBC consults Fraud after its own IP/barring playbook and before trunk reservation (fail-closed), while Switching propagates `account_id`; alerts are stored and published to `fraud.alerts` without blocking calls.

**Tech Stack:** Go 1.27.4, protobuf/gRPC, Kafka (franz-go), Postgres/pgx, OpenTelemetry, Docker Compose, stdlib CLI/HTTP

## Global Constraints

- Source: `docs/superpowers/specs/2026-11-18-fraud-mvp-design.md` (P6 first slice). Work in place on the current checkout.
- Generated protobuf code is produced only by `make protoc` and committed. No manual edits to generated protobuf files.
- Every task ends with `go vet ./...` in every touched Go module and a lowercase conventional commit. Commit messages must remain exactly as written in each task.
- Fraud DB tests require the dev stack and use `FRAUD_DSN` with fallback `postgres://postgres:dev-only@localhost:5432/fraud`. SBC tests require the dev stack and continue using `SBC_DSN` with fallback `postgres://postgres:dev-only@localhost:5432/sbc`.
- Full gate after all tasks: rebuild changed images, `make up`, `make smoke`, then `make test`.
- No call blocking from the advisory score. Only explicit prefix/CLI blocks and velocity caps deny.
- Fraud never returns raw DB errors over gRPC.
- Alert publish failure never rolls back a DB alert.
- Kafka offsets commit only after successful CDR DB transactions.
- No placeholders/TBD/TODO.

```go
package scoring

import "time"

const RuleVersion = "v1"

type Config struct {
	AlertThreshold   int
	SoftAccountCalls int
	SoftPrefixCalls  int
}

type Signals struct {
	Destination      string
	CLI              string
	Start            time.Time
	RiskPrefixWeight int
	FirstSeenPrefix  bool
	FirstSeenCLI     bool
	AccountCalls     int
	PrefixCalls      int
}

func DestinationPrefix(destination string) string {
	if len(destination) <= 4 {
		return destination
	}
	return destination[:4]
}

func Score(cfg Config, s Signals) (int, []string) {
	score := 0
	reasons := []string{}
	if s.RiskPrefixWeight > 0 {
		score += s.RiskPrefixWeight
		reasons = append(reasons, "risk_prefix")
	}
	if s.FirstSeenPrefix {
		score += 20
		reasons = append(reasons, "first_seen_prefix")
	}
	if cfg.SoftAccountCalls > 0 && s.AccountCalls >= cfg.SoftAccountCalls ||
		cfg.SoftPrefixCalls > 0 && s.PrefixCalls >= cfg.SoftPrefixCalls {
		score += 20
		reasons = append(reasons, "velocity")
	}
	if s.FirstSeenCLI {
		score += 15
		reasons = append(reasons, "first_seen_cli")
	}
	if s.Start.UTC().Hour() < 6 {
		score += 10
		reasons = append(reasons, "odd_hour")
	}
	if score > 100 {
		score = 100
	}
	return score, reasons
}

func Level(score int) string {
	switch {
	case score >= 80:
		return "high"
	case score >= 40:
		return "medium"
	default:
		return "low"
	}
}
```

```go
package main

import (
	"context"
	_ "embed"
	"encoding/json"
	"errors"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgtype"
	"github.com/jackc/pgx/v5/pgxpool"

	fraudpb "omnifoni/fraud/gen/fraudpb"
)

//go:embed migrate.sql
var schemaSQL string

type Store struct {
	pool *pgxpool.Pool
}

func Open(ctx context.Context, dsn string) (*Store, error) {
	pool, err := pgxpool.New(ctx, dsn)
	if err != nil {
		return nil, err
	}
	return &Store{pool: pool}, nil
}

func (s *Store) Close() {
	s.pool.Close()
}

func (s *Store) Migrate(ctx context.Context) error {
	_, err := s.pool.Exec(ctx, schemaSQL)
	return err
}

func (s *Store) Reserve(ctx context.Context, req *fraudpb.CheckCallRequest, cfg Config) (*fraudpb.CheckCallResponse, *fraudpb.Alert, error) {
	account := req.GetAccountId()
	destination := req.GetDestination()
	prefix := DestinationPrefix(destination)
	day := time.Now().UTC()

	tx, err := s.pool.Begin(ctx)
	if err != nil {
		return nil, nil, err
	}
	defer tx.Rollback(ctx)

	if _, err := tx.Exec(ctx,
		`SELECT pg_advisory_xact_lock(hashtext('fraud:' || $1))`, account); err != nil {
		return nil, nil, err
	}

	enabled := true
	var maxDay, maxPrefix pgtype.Int4
	err = tx.QueryRow(ctx,
		`SELECT enabled, max_calls_per_day, max_calls_per_prefix_day
		 FROM fraud_controls WHERE account_id = $1`, account).
		Scan(&enabled, &maxDay, &maxPrefix)
	if errors.Is(err, pgx.ErrNoRows) {
		enabled = true
	} else if err != nil {
		return nil, nil, err
	}
	if !enabled {
		maxDay = pgtype.Int4{}
		maxPrefix = pgtype.Int4{}
	}

	blockedDestination, err := prefixBlocked(ctx, tx, "prefix_blocks", account, destination)
	if err != nil {
		return nil, nil, err
	}
	blockedCLI, err := prefixBlocked(ctx, tx, "cli_blocks", account, req.GetCli())
	if err != nil {
		return nil, nil, err
	}
	signals, err := readSignals(ctx, tx, account, destination, prefix, req.GetCli(), cfg, day)
	if err != nil {
		return nil, nil, err
	}
	score, reasons := Score(cfg, signals)
	resp := &fraudpb.CheckCallResponse{
		Allowed:     true,
		RiskScore:   int32(score),
		RiskLevel:   Level(score),
		Reasons:     reasons,
		RuleVersion: RuleVersion,
	}

	var alreadyReserved bool
	if err := tx.QueryRow(ctx,
		`SELECT EXISTS(SELECT 1 FROM reservations WHERE call_id = $1)`, req.GetCallId()).
		Scan(&alreadyReserved); err != nil {
		return nil, nil, err
	}
	if alreadyReserved {
		resp.Allowed = true
		resp.Reason = ""
		if err := tx.Commit(ctx); err != nil {
			return nil, nil, err
		}
		return resp, nil, nil
	}

	reason := hardRule(blockedDestination, blockedCLI, maxDay, maxPrefix,
		signals.AccountCalls, signals.PrefixCalls)
	if reason != "" {
		resp.Allowed = false
		resp.Reason = reason
	}

	var alert *fraudpb.Alert
	if score >= cfg.AlertThreshold {
		var alreadyAlerted bool
		if err := tx.QueryRow(ctx,
			`SELECT EXISTS(SELECT 1 FROM alerts WHERE call_id = $1)`, req.GetCallId()).
			Scan(&alreadyAlerted); err != nil {
			return nil, nil, err
		}
		if !alreadyAlerted {
			alert, err = insertAlert(ctx, tx, req, score, resp.GetRiskLevel(), reasons)
			if err != nil {
				return nil, nil, err
			}
		}
	}

	if reason == "" {
		if _, err := tx.Exec(ctx,
			`INSERT INTO reservations(call_id, account_id, day, prefix)
			 VALUES ($1, $2, $3, $4)`, req.GetCallId(), account, day, prefix); err != nil {
			return nil, nil, err
		}
		for _, p := range []string{"", prefix} {
			if _, err := tx.Exec(ctx,
				`INSERT INTO daily_usage(account_id, day, prefix, active)
				 VALUES ($1, $2, $3, 1)
				 ON CONFLICT (account_id, day, prefix)
				 DO UPDATE SET active = daily_usage.active + 1`, account, day, p); err != nil {
				return nil, nil, err
			}
		}
	}
	if err := tx.Commit(ctx); err != nil {
		return nil, nil, err
	}
	return resp, alert, nil
}

func (s *Store) EndCall(ctx context.Context, callID string) error {
	tx, err := s.pool.Begin(ctx)
	if err != nil {
		return err
	}
	defer tx.Rollback(ctx)

	var account, prefix string
	var day time.Time
	err = tx.QueryRow(ctx,
		`DELETE FROM reservations WHERE call_id = $1
		 RETURNING account_id, day, prefix`, callID).Scan(&account, &day, &prefix)
	if errors.Is(err, pgx.ErrNoRows) {
		return tx.Commit(ctx)
	}
	if err != nil {
		return err
	}
	if err := decrementActive(ctx, tx, account, day, prefix); err != nil {
		return err
	}
	return tx.Commit(ctx)
}

func decrementActive(ctx context.Context, tx pgx.Tx, account string, day time.Time, prefix string) error {
	prefixes := []string{""}
	if prefix != "" {
		prefixes = append(prefixes, prefix)
	}
	for _, p := range prefixes {
		if _, err := tx.Exec(ctx,
			`UPDATE daily_usage SET active = GREATEST(active - 1, 0)
			 WHERE account_id = $1 AND day = $2 AND prefix = $3`, account, day, p); err != nil {
			return err
		}
	}
	return nil
}

func hardRule(blockedDestination, blockedCLI bool, maxDay, maxPrefix pgtype.Int4, accountCalls, prefixCalls int) string {
	switch {
	case blockedDestination:
		return "fraud_destination"
	case blockedCLI:
		return "fraud_cli"

		return "fraud_velocity_account"
	case maxPrefix.Valid && prefixCalls >= int(maxPrefix.Int32):
		return "fraud_velocity_prefix"
	default:
		return ""
	}
}

func prefixBlocked(ctx context.Context, tx pgx.Tx, table, account, value string) (bool, error) {
	if value == "" {
		return false, nil
	}
	query := `SELECT prefix FROM prefix_blocks
	          WHERE account_id = $1 AND $2 LIKE prefix || '%'
	          ORDER BY length(prefix) DESC LIMIT 1`

		         WHERE account_id = $1 AND $2 LIKE prefix || '%'

	if errors.Is(err, pgx.ErrNoRows) {

	}

	}

}

func readSignals(ctx context.Context, tx pgx.Tx, account, destination, prefix, cli string, cfg Config, day time.Time) (Signals, error) {
	s := Signals{
		Destination:      destination,
		CLI:              cli,
		Start:            time.Now().UTC(),
		RiskPrefixWeight: riskPrefixWeight(ctx, tx, destination),
	}

	rows, err := tx.Query(ctx,
		`SELECT destination FROM processed_cdrs WHERE account_id = $1`, account)
	if err != nil {
		return Signals{}, err
	}
	defer rows.Close()
	for rows.Next() {
		var seenDestination string
		if err := rows.Scan(&seenDestination); err != nil {
			return Signals{}, err
		}
		if DestinationPrefix(seenDestination) == prefix {
			s.FirstSeenPrefix = true
			break
		}
	}
	if err := rows.Err(); err != nil {
		return Signals{}, err
	}
	s.FirstSeenPrefix = !s.FirstSeenPrefix

	err = tx.QueryRow(ctx,
		`SELECT EXISTS(SELECT 1 FROM processed_cdrs WHERE account_id = $1 AND cli = $2)`,
		account, cli).Scan(&s.FirstSeenCLI)
	if err != nil {
		return Signals{}, err
	}
	s.FirstSeenCLI = !s.FirstSeenCLI

	if err := tx.QueryRow(ctx,
		`SELECT COALESCE(SUM(active + completed), 0)
		 FROM daily_usage WHERE account_id = $1 AND day = $2::date AND prefix = ''`,
		account, day).
		Scan(&s.AccountCalls); err != nil {
		return Signals{}, err
	}
	if err := tx.QueryRow(ctx,
		`SELECT COALESCE(SUM(active + completed), 0)
		 FROM daily_usage WHERE account_id = $1 AND day = $3::date AND prefix = $2`,
		account, prefix, day).
		Scan(&s.PrefixCalls); err != nil {
		return Signals{}, err
	}
	_ = cfg
	return s, nil
}

func riskPrefixWeight(ctx context.Context, tx pgx.Tx, destination string) int {
	var weight int
	err := tx.QueryRow(ctx,
		`SELECT weight FROM risk_prefixes
		 WHERE $1 LIKE prefix || '%'
		 ORDER BY length(prefix) DESC
		 LIMIT 1`, destination).Scan(&weight)
	if errors.Is(err, pgx.ErrNoRows) {
		return 0
	}
	if err != nil {
		return 0
	}
	return weight
}
```

```go
func (s *Store) ListAlerts(ctx context.Context, accountID string, limit int, cursor string) ([]*fraudpb.Alert, string, error) {
	if limit <= 0 {
		limit = 100
	}
	if limit > 1000 {
		limit = 1000
	}

	query := `SELECT alert_id::text, call_id, account_id, risk_score, risk_level,
	                 reasons, rule_version, created_at, status
	          FROM alerts
	          WHERE ($1 = '' OR account_id = $1)
	          ORDER BY created_at DESC, alert_id DESC
	          LIMIT $2`
	args := []any{accountID, limit}
	if cursor != "" {
		createdAt, alertID, err := decodeAlertCursor(cursor)
		if err != nil {
			return nil, "", err
		}
		query = `SELECT alert_id::text, call_id, account_id, risk_score, risk_level,
		                reasons, rule_version, created_at, status
		         FROM alerts
		         WHERE ($1 = '' OR account_id = $1)
		           AND (created_at, alert_id) < ($2::timestamptz, $3::uuid)
		         ORDER BY created_at DESC, alert_id DESC
		         LIMIT $4`
		args = []any{accountID, createdAt, alertID, limit}
	}

	rows, err := s.pool.Query(ctx, query, args...)
	if err != nil {
		return nil, "", err
	}
	defer rows.Close()

	alerts := []*fraudpb.Alert{}
	for rows.Next() {
		var alertID, callID, account, level, version, status string
		var score int
		var rawReasons []byte
		var createdAt time.Time
		if err := rows.Scan(&alertID, &callID, &account, &score, &level,
			&rawReasons, &version, &createdAt, &status); err != nil {
			return nil, "", err
		}
		reasons := []string{}
		if err := json.Unmarshal(rawReasons, &reasons); err != nil {
			return nil, "", err
		}
		alerts = append(alerts, &fraudpb.Alert{
			AlertId:     alertID,
			CallId:      callID,
			AccountId:   account,
			RiskScore:   int32(score),
			RiskLevel:   level,
			Reasons:     reasons,
			RuleVersion: version,
			CreatedAtMs: createdAt.UnixMilli(),
			Status:      status,
		})
	}
	if err := rows.Err(); err != nil {
		return nil, "", err
	}
	next := ""
	if len(alerts) == limit {
		last := alerts[len(alerts)-1]
		next = encodeAlertCursor(last)
	}
	return alerts, next, nil
}

func decodeAlertCursor(cursor string) (time.Time, string, error) {
	raw, err := base64.RawURLEncoding.DecodeString(cursor)
	if err != nil {
		return time.Time{}, "", err
	}
	parts := strings.SplitN(string(raw), "|", 2)
	if len(parts) != 2 || parts[1] == "" {
		return time.Time{}, "", errors.New("invalid alert cursor")
	}
	ms, err := strconv.ParseInt(parts[0], 10, 64)
	if err != nil {
		return time.Time{}, "", err
	}
	return time.UnixMilli(ms).UTC(), parts[1], nil
}

func encodeAlertCursor(alert *fraudpb.Alert) string {
	raw := fmt.Sprintf("%d|%s", alert.GetCreatedAtMs(), alert.GetAlertId())
	return base64.RawURLEncoding.EncodeToString([]byte(raw))
}
```

```go
type FraudServer struct {

	cfg       Config
	publisher Publisher

func (s *FraudServer) CheckCall(ctx context.Context, req *fraudpb.CheckCallRequest) (*fraudpb.CheckCallResponse, error) {
	ctx, span := otel.Tracer("omnifoni/fraud").Start(ctx, "fraud.check_call",
		trace.WithAttributes(
			attribute.String("account_id", req.GetAccountId()),
			attribute.String("destination", observability.Mask(req.GetDestination()))))
	defer span.End()

	if req.GetCallId() == "" || req.GetAccountId() == "" {
		return nil, status.Error(codes.InvalidArgument, "call_id and account_id are required")
	}
	resp, alert, err := s.store.Reserve(ctx, req, s.cfg)
	if err != nil {
		slog.ErrorContext(ctx, "fraud reserve failed", "err", err)
		span.RecordError(err)
		return nil, status.Error(codes.Internal, "fraud store failure")
	}
	if alert != nil && s.publisher != nil {
		event := &fraudpb.FraudAlert{EventId: uuid.NewString(), Alert: alert}
		if err := s.publisher.Publish(ctx, event); err != nil {
			slog.ErrorContext(ctx, "fraud alert publish failed", "err", err)
			span.RecordError(err)
		}
	}
	return resp, nil
}

func (s *FraudServer) EndCall(ctx context.Context, req *fraudpb.EndCallRequest) (*fraudpb.EndCallResponse, error) {
	ctx, span := otel.Tracer("omnifoni/fraud").Start(ctx, "fraud.end_call",
		trace.WithAttributes(attribute.String("call_id", req.GetCallId())))
	defer span.End()

	if req.GetCallId() == "" {
		return nil, status.Error(codes.InvalidArgument, "call_id is required")
	}
	if err := s.store.EndCall(ctx, req.GetCallId()); err != nil {
		slog.ErrorContext(ctx, "fraud end call failed", "err", err)
		span.RecordError(err)
		return nil, status.Error(codes.Internal, "fraud store failure")
	}
	return &fraudpb.EndCallResponse{}, nil
}

		return nil, status.Error(codes.Internal, "fraud store failure")
	}
	return &fraudpb.ListAlertsResponse{Alerts: alerts, NextCursor: next}, nil
}
```

```go
func (s *Store) IngestCDR(ctx context.Context, c *mediationpb.CallCdr) error {
	account := c.GetAccountId()
	prefix := DestinationPrefix(c.GetDestination())

	tx, err := s.pool.Begin(ctx)
	if err != nil {
		return err
	}
	defer tx.Rollback(ctx)

	if _, err := tx.Exec(ctx,
		`SELECT pg_advisory_xact_lock(hashtext('fraud:' || $1))`, account); err != nil {
		return err
	}

	var seen bool
	if err := tx.QueryRow(ctx,
		`SELECT EXISTS(SELECT 1 FROM processed_cdrs WHERE call_id = $1)`, c.GetCallId()).
		Scan(&seen); err != nil {
		return err
	}
	if seen {
		return tx.Commit(ctx)
	}

	usageAccount := account
	usageDay := time.Now().UTC()
	usagePrefix := prefix

	var resAccount, resPrefix string
	var resDay time.Time
	err = tx.QueryRow(ctx,
		`DELETE FROM reservations WHERE call_id = $1
		 RETURNING account_id, day, prefix`, c.GetCallId()).
		Scan(&resAccount, &resDay, &resPrefix)
	if errors.Is(err, pgx.ErrNoRows) {
		err = nil
	} else if err == nil {
		usageAccount = resAccount
		usageDay = resDay
		usagePrefix = resPrefix
		if err := decrementActive(ctx, tx, resAccount, resDay, resPrefix); err != nil {
			return err
		}
	}
	if err != nil {
		return err
	}

	for _, p := range []string{"", usagePrefix} {
		if _, err := tx.Exec(ctx,
			`INSERT INTO daily_usage(account_id, day, prefix, completed, seconds, amount_minor)
			 VALUES ($1, $2, $3, 1, $4, $5)
			 ON CONFLICT (account_id, day, prefix) DO UPDATE SET
			   completed = daily_usage.completed + 1,
			   seconds = daily_usage.seconds + EXCLUDED.seconds,
			   amount_minor = daily_usage.amount_minor + EXCLUDED.amount_minor`,
			usageAccount, usageDay, p, c.GetChargedSeconds(), c.GetAmountMinor()); err != nil {
			return err
		}
	}
	if _, err := tx.Exec(ctx,
		`INSERT INTO processed_cdrs(call_id, account_id, destination, cli, seconds, amount_minor)
		 VALUES ($1, $2, $3, $4, $5, $6)`,
		c.GetCallId(), account, c.GetDestination(), c.GetCli(),
		c.GetChargedSeconds(), c.GetAmountMinor()); err != nil {
		return err
	}
	return tx.Commit(ctx)
}
```

```go
package main

import (
	"context"
	"log/slog"
	"strconv"
	"time"

	kgo "github.com/twmb/franz-go/pkg/kgo"
	"google.golang.org/protobuf/proto"

	mediationpb "omnifoni/mediation/gen/mediationpb"
	"omnifoni/observability"
)

type Handler interface {
	Handle(context.Context, *kgo.Record) error
}

func Run(ctx context.Context, cl *kgo.Client, h Handler) {
	for {
		fetches := cl.PollRecords(ctx, 100)
		fetches.EachError(func(topic string, partition int32, err error) {
			slog.ErrorContext(ctx, "kafka fetch", "topic", topic, "partition", partition, "err", err)
		})
		lasts := map[string]*kgo.Record{}
		for iter := fetches.RecordIter(); !iter.Done(); {
			rec := iter.Next()
			for {
				err := h.Handle(ctx, rec)
				if err == nil {
					break
				}
				if ctx.Err() != nil {
					return
				}
				slog.ErrorContext(ctx, "fraud handle failed, retrying", "err", err)
				select {
				case <-ctx.Done():
					return
				case <-time.After(time.Second):
				}
			}
			lasts[rec.Topic+"/"+strconv.Itoa(int(rec.Partition))] = rec
		}
		if ctx.Err() != nil {
			return
		}
		toCommit := make([]*kgo.Record, 0, len(lasts))
		for _, rec := range lasts {
			toCommit = append(toCommit, rec)
		}
		if len(toCommit) > 0 {
			if err := cl.CommitRecords(ctx, toCommit...); err != nil && ctx.Err() == nil {
				slog.ErrorContext(ctx, "fraud offset commit failed", "err", err)
			}
		}
	}
}

type Ingester struct {
	store *Store
}

func NewIngester(store *Store) *Ingester {
	return &Ingester{store: store}
}

func (i *Ingester) Handle(ctx context.Context, rec *kgo.Record) error {
	c := &mediationpb.CallCdr{}
	if err := proto.Unmarshal(rec.Value, c); err != nil {
		slog.ErrorContext(ctx, "fraud cdr decode failed", "err", err)
		return nil
	}
	if c.GetCallId() == "" || c.GetAccountId() == "" {
		slog.ErrorContext(ctx, "fraud cdr validation failed", "call_id", c.GetCallId())
		return nil
	}
	ctx = observability.ExtractKafka(ctx, rec)
	ctx, span := observability.ConsumerSpan(ctx, "fraud.consume_cdr", c.GetCallId())
	defer span.End()
	if err := i.store.IngestCDR(ctx, c); err != nil {
		span.RecordError(err)
		return err
	}
	return nil
}
```

```go
type storeHandler struct {
	store *Store
	mu    sync.Mutex
	order []string
	fail  map[string]int
}

func (h *storeHandler) Handle(ctx context.Context, rec *kgo.Record) error {
	c := &mediationpb.CallCdr{}
	if err := proto.Unmarshal(rec.Value, c); err != nil {
		return err
	}
	h.mu.Lock()
	h.order = append(h.order, c.GetCallId())
	shouldFail := h.fail[c.GetCallId()] == 0
	if shouldFail {
		h.fail[c.GetCallId()]++
	}
	h.mu.Unlock()
	if shouldFail {
		return errors.New("transient store failure")
	}
	return h.store.IngestCDR(ctx, c)
}

					cancel()
					return
				}
			}
		}
		time.Sleep(200 * time.Millisecond)
	}
	t.Fatal("two CDRs were not processed and committed")
}
```

```go
import (

	fraudpb "omnifoni/fraud/gen/fraudpb"
	"omnifoni/observability"
)

type kafkaPublisher struct {
	prod  *kgo.Client
	topic string
}

func (p *kafkaPublisher) Publish(ctx context.Context, event *fraudpb.FraudAlert) error {
	raw, err := proto.Marshal(event)
	if err != nil {
		return err
	}
	rec := &kgo.Record{
		Topic: p.topic,
		Key:   []byte(event.GetAlert().GetAccountId()),
		Value: raw,
	}
	observability.InjectKafka(ctx, rec)
	return p.prod.ProduceSync(ctx, rec).FirstErr()
}

func env(key, def string) string {
	if value := os.Getenv(key); value != "" {
		return value
	}
	return def
}

func envInt(key string, def int) int {
	value := os.Getenv(key)
	if value == "" {
		return def
	}
	n, err := strconv.Atoi(value)
	if err != nil || n < 1 {
		slog.Error("invalid integer env", "key", key, "value", value)
		os.Exit(1)
	}
	return n
}

func main() {
	observability.SetupLogging("fraud")
	shutdown, err := observability.Init(context.Background(), "fraud")
	if err != nil {
		slog.Warn("tracing disabled", "err", err)
	}
	defer func() { _ = shutdown(context.Background()) }()

	addr := env("FRAUD_ADDR", ":50063")
	dsn := env("FRAUD_DSN", "postgres://postgres:dev-only@postgres:5432/fraud")
	brokers := env("FRAUD_KAFKA", "kafka:9092")
	alertsTopic := env("FRAUD_TOPIC_ALERTS", "fraud.alerts")
	cfg := Config{
		AlertThreshold:   envInt("FRAUD_ALERT_THRESHOLD", 60),
		SoftAccountCalls: envInt("FRAUD_SOFT_ACCOUNT_CALLS", 5),
		SoftPrefixCalls:  envInt("FRAUD_SOFT_PREFIX_CALLS", 2),
	}

	ctx, cancel := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
	defer cancel()

	store, err := Open(ctx, dsn)
	if err != nil {
		slog.Error("db pool", "err", err)
		os.Exit(1)
	}
	defer store.Close()
	if err := store.Migrate(ctx); err != nil {
		slog.Error("migrate", "err", err)
		os.Exit(1)
	}

	consumer, err := kgo.NewClient(
		kgo.SeedBrokers(brokers),
		kgo.ConsumerGroup("fraud"),
		kgo.ConsumeTopics("cdrs.normalized"),
		kgo.ConsumeStartOffset(kgo.NewOffset().AtStart()),
		kgo.DisableAutoCommit(),
		kgo.AllowAutoTopicCreation(),
	)
	if err != nil {
		slog.Error("kafka consumer", "err", err)
		os.Exit(1)
	}
	defer consumer.Close()

	producer, err := kgo.NewClient(
		kgo.SeedBrokers(brokers),
		kgo.AllowAutoTopicCreation(),
	)
	if err != nil {
		slog.Error("kafka producer", "err", err)
		os.Exit(1)
	}
	defer producer.Close()

	publisher := &kafkaPublisher{prod: producer, topic: alertsTopic}
	server := NewFraudServer(store, cfg, publisher)
	ingester := NewIngester(store)
	go Run(ctx, consumer, ingester)

	lis, err := net.Listen("tcp", addr)
	if err != nil {
		slog.Error("listen", "err", err)
		os.Exit(1)
	}
	g := grpc.NewServer(observability.ServerOptions()...)
	fraudpb.RegisterFraudServer(g, server)
	if env("FRAUD_REFLECTION", "") == "1" {
		reflection.Register(g)
	}
	slog.Info("fraud up", "addr", addr)
	if err := g.Serve(lis); err != nil {
		slog.Error("serve failed", "err", err)
		os.Exit(1)
	}
}
```

```go
package main

import (
	"bufio"
	"encoding/json"
	"flag"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"time"

	"github.com/google/uuid"

	fraudscore "omnifoni/fraud/scoring"
)

type history struct {
	Destination string `json:"destination"`
	CLI         string `json:"cli"`
	StartedAt   string `json:"started_at"`
}

type testCase struct {
	AccountID   string    `json:"account_id"`
	Destination string    `json:"destination"`
	CLI         string    `json:"cli"`
	Start       string    `json:"start"`
	History     []history `json:"history"`
	Fraud       bool      `json:"fraud"`
}

type result struct {
	TP int
	FP int
	FN int
	TN int
}

func riskWeight(destination string) int {
	if strings.HasPrefix(destination, "900") {
		return 30
	}
	return 0
}

func signalsFor(c testCase, cfg fraudscore.Config, start time.Time) fraudscore.Signals {
	prefix := fraudscore.DestinationPrefix(c.Destination)
	seenPrefix := false
	firstCLI := true
	accountCalls := 0
	prefixCalls := 0
	for _, h := range c.History {
		if fraudscore.DestinationPrefix(h.Destination) == prefix {
			prefixCalls++
			seenPrefix = true
		}
		if h.CLI == c.CLI {
			firstCLI = false
		}
		if h.StartedAt != "" {
			accountCalls++
		}
	}
	return fraudscore.Signals{
		Destination:      c.Destination,
		CLI:              c.CLI,
		Start:            start,
		RiskPrefixWeight: riskWeight(c.Destination),
		FirstSeenPrefix:  !seenPrefix,
		FirstSeenCLI:     firstCLI,
		AccountCalls:     accountCalls,
		PrefixCalls:      prefixCalls,
	}
}

func run(path string, cfg fraudscore.Config) (result, int, error) {
	f, err := os.Open(path)
	if err != nil {
		return result{}, 0, err
	}
	defer f.Close()

	res := result{}
	count := 0
	scanner := bufio.NewScanner(f)
	for scanner.Scan() {
		line := strings.TrimSpace(scanner.Text())
		if line == "" {
			continue
		}
		c := testCase{}
		if err := json.Unmarshal([]byte(line), &c); err != nil {
			return result{}, 0, err
		}
		start, err := time.Parse(time.RFC3339, c.Start)
		if err != nil {
			return result{}, 0, err
		}
		score, _ := fraudscore.Score(cfg, signalsFor(c, cfg, start))
		predicted := score >= cfg.AlertThreshold
		count++
		switch {
		case predicted && c.Fraud:
			res.TP++
		case predicted && !c.Fraud:
			res.FP++
		case !predicted && c.Fraud:
			res.FN++
		default:
			res.TN++
		}
	}
	if err := scanner.Err(); err != nil {
		return result{}, 0, err
	}
	return res, count, nil
}

func ratio(num, den int) float64 {
	if den == 0 {
		return 1.0
	}
	return float64(num) / float64(den)
}

func report(res result, count int, cfg fraudscore.Config) string {
	precision := ratio(res.TP, res.TP+res.FP)

precision: %.3f
recall: %.3f

}

		fraudscore.RuleVersion, count, res.TP, res.FP, res.FN, res.TN, precision, recall, uuid.NewString())
```

```go
package main

import (
	"context"
	"fmt"
	"os"
	"time"

	"github.com/google/uuid"
	"google.golang.org/grpc"
	"google.golang.org/grpc/credentials/insecure"

	fraudpb "omnifoni/fraud/gen/fraudpb"
)

func env(key, def string) string {
	if value := os.Getenv(key); value != "" {
		return value
	}
	return def
}

func fail(format string, args ...any) {
	fmt.Fprintf(os.Stderr, format+"\n", args...)
	os.Exit(1)
}

func main() {
	addr := env("FRAUD_ADDR", "localhost:50063")
	account := env("FRAUD_SMOKE_ACCOUNT", "acc-smoke-fraud")
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()

	conn, err := grpc.NewClient(addr, grpc.WithTransportCredentials(insecure.NewCredentials()))
	if err != nil {
		fail("dial fraud: %v", err)
	}
	defer conn.Close()
	client := fraudpb.NewFraudClient(conn)

	firstID := "smoke-fraud-" + uuid.NewString()
	first, err := client.CheckCall(ctx, &fraudpb.CheckCallRequest{
		CallId: firstID, AccountId: account,
		Destination: "9001234", Cli: "27820000000",
	})
	if err != nil {
		fail("first check: %v", err)
	}
	if !first.GetAllowed() {
		fail("first check denied: %s", first.GetReason())
	}

	secondID := "smoke-fraud-" + uuid.NewString()
	second, err := client.CheckCall(ctx, &fraudpb.CheckCallRequest{
		CallId: secondID, AccountId: account,
		Destination: "9001234", Cli: "27820000000",
	})
	if err != nil {
		fail("second check: %v", err)
	}
	if second.GetAllowed() || second.GetReason() != "fraud_velocity_account" {
		fail("want fraud_velocity_account, got %+v", second)
	}

	alerts, err := client.ListAlerts(ctx, &fraudpb.ListAlertsRequest{AccountId: account, Limit: 10})
	if err != nil {
		fail("list alerts: %v", err)
	}
	if len(alerts.GetAlerts()) == 0 {
		fail("no fraud alerts for %s", account)
	}

	if _, err := client.EndCall(ctx, &fraudpb.EndCallRequest{CallId: firstID}); err != nil {
		fail("end first call: %v", err)
	}
	fmt.Println("ok: fraud")
}
```
